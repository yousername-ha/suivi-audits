function analyze(it){
  const N=S.nextYear, P=N-1, r=[]; let sc=0;
  const fs=S.findings.filter(f=>f.items.includes(it.id));
  const open=fs.filter(f=>f.status!=='Clos'), closed=fs.filter(f=>f.status==='Clos');
  const socle=it.ref!=='A';
  if(socle) r.push('Socle audité à chaque visite');
  for(const f of open){
    if(f.type.startsWith('NC')){sc+=100;r.push(`${typeLabel(f.type)} ouverte (${f.source} ${f.year}) : sa levée sera vérifiée`)}
    else if(f.type==='PS'&&f.source==='AFNOR'){sc+=70;r.push(`Point sensible relevé par l'AFNOR en ${f.year}`)}
    else if(f.type==='PS'){sc+=35;r.push(`Point sensible de l'audit interne ${f.year}`)}
    else {sc+=10;r.push(`Remarque ${f.source} ${f.year} encore ouverte`)}
  }
  if(closed.length){sc+=15;r.push(closed.length>1?`${closed.length} écarts clos : preuves de clôture à présenter`:'Écart clos : preuves de clôture à présenter')}
  let blind=false, notSeen=false;
  const hasP=S.years.includes(P);
  if(!socle&&hasP){
    if(!it.afnor[P]){sc+=60;notSeen=true;r.push(`Non vue par l'AFNOR en ${P} : rotation du cycle de surveillance`)}
    if(!it.afnor[P]&&!it.interne[P]){sc+=25;blind=true;r.push(`Angle mort ${P} : ni audit interne ni AFNOR`)}
  }
  if(!socle&&it.afnor[N]){sc+=40;r.push(`Inscrite au plan d'audit AFNOR ${N}`)}
  let level;
  if(socle) level=open.length?'certain':'socle';
  else level=sc>=100?'certain':sc>=60?'tres':sc>=30?'probable':'faible';
  const internalGap=(level==='certain'||level==='tres')&&!it.interne[N]&&S.years.includes(N);
  return {sc,level,reasons:r,open,closed,blind,notSeen,internalGap,fs};
}

function indState(ind){
  const auto=ind.calc&&ind.auto!==false?indCalc(ind):null;
  const all=(ind.values||[]).slice().sort((a,b)=>String(a.date).localeCompare(String(b.date)));
  const vals=all.filter(x=>!x.hist), hist=all.filter(x=>x.hist);
  const last=vals[vals.length-1], prev=vals[vals.length-2]||hist[hist.length-1];
  const v=auto?auto.v:(last?last.v:null);
  let st='na';
  if(!ind.dir||ind.target===''||ind.target==null)st='undef';
  else if(v!=null&&v!==''){
    if(ind.dir==='=')st=String(v).toLowerCase()===String(ind.target).toLowerCase()?'ok':'ko';
    else if(ind.dir==='≤')st=+v<=+ind.target?'ok':'ko';
    else if(ind.dir==='≥')st=+v>=+ind.target?'ok':'ko';
  }
  let trend='';
  if(!auto&&last&&prev&&typeof last.v==='number'&&typeof prev.v==='number')trend=last.v>prev.v?'↗':last.v<prev.v?'↘':'→';
  const h=hist[hist.length-1];
  return {v,st,trend,auto,date:auto?'calculé':last?last.date:'',hist:h?{per:h.hist,v:h.v}:null};
}
